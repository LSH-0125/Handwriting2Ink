# --- OCR ---
# These versions MUST stay pinned.
# paddlex >= 3.2.0 requires modelscope unconditionally, and modelscope
# imports torch at import time. That chain
#   paddleocr -> paddlex -> modelscope -> torch
# bloats the image and makes "import paddleocr" fail outright when the
# torch install is not perfect. paddlex <= 3.1.0 is clean.
# paddleocr only constrains paddlex with ">=", so pinning paddleocr alone
# still resolves to the newest paddlex. Pin both.
# Verified with a pip dry-run: 101 packages, no torch / modelscope / nvidia-*.
paddleocr==3.1.0
paddlex[ie,multimodal,ocr,trans]==3.1.0
paddlepaddle==3.0.0

# paddlex 3.1.0 does "from langchain.docstore.document import Document",
# a module that langchain 1.x removed. paddlex declares langchain with no
# upper bound, so a fresh install picks 1.x and then
#   import paddleocr -> ModuleNotFoundError: No module named 'langchain.docstore'
# The server still starts; every OCR job fails instead. Verified locally:
# with langchain<1 the import succeeds.
langchain<1

# paddlex requires opencv-contrib-python. Installing opencv-python-headless
# alongside it makes two distributions fight over the same cv2 module, so
# keep only one. Non-headless needs libGL, installed in the Dockerfile.
opencv-contrib-python==4.10.0.84
